cmake_minimum_required(VERSION 3.22)
project(uhf_v3_core LANGUAGES CXX)
include(CTest)
option(UHF_V3_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
add_library(uhf_v3_core STATIC protocol.cpp ../domain/modbus.cpp)
target_compile_features(uhf_v3_core PUBLIC cxx_std_17)
target_include_directories(uhf_v3_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(uhf_v3_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)
    if(UHF_V3_SANITIZERS)
        target_compile_options(uhf_v3_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(uhf_v3_core PUBLIC -fsanitize=address,undefined)
    endif()
endif()
if(BUILD_TESTING)
    add_executable(v3_protocol_test ../../tests/v3/protocol_test.cpp)
    target_link_libraries(v3_protocol_test PRIVATE uhf_v3_core)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(v3_protocol_test PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)
    endif()
    add_test(NAME v3_protocol COMMAND v3_protocol_test)
endif()
